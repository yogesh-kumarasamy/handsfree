# Routines Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Per-project scheduled agent runs (cron + prompt + agent command) whose final text is posted to the project's collab channel; two presets (daily PR check, Sentry check 3×/day); Run now, last result and a log per routine.

**Architecture:** A pure cron module (`engine/src/routines/cron.ts`) and a `RoutineService` (`engine/src/routines/service.ts`) with a 30-second tick that runs due routines through the shell (`sh`), parses Claude's JSON envelope with `lastJson`, records `RoutineState` in the engine state, and posts through a new `MessagingService.postToProject`. The orchestrator owns the service; two routes expose Run now and the log; the Projects editor gets a Routines step and the Dashboard a Routines card.

**Tech Stack:** Node 20, TypeScript/esbuild, `node --test`, React 18, existing e2e harness and fakes.

**Spec:** `docs/superpowers/specs/2026-10-15-routines-design.md`

## Global Constraints

- No new runtime dependencies.
- Cron is 5-field local time; invalid expressions never throw at runtime (routine stays unscheduled, UI shows the error).
- At most 2 routine runs at once across projects; never two runs of the same routine.
- Default command `claude -p {prompt} --output-format json --max-turns 25 --dangerously-skip-permissions`, cwd = the project's `repoPath`, timeout `timeoutMin` (1..120, default 15).
- Posting goes through `MessagingService.postToProject(p, text, "routine", idempotencyKey)`; a post failure never fails the run.
- Routine logs: `LOG_DIR/routine-<id>.log` (sanitised id).
- **Parallel work notice:** `worktree-trackers` is being built at the same time and owns the tracker/OAuth code, the Projects "Tracker" step (index 1), Settings Accounts, dashboard/sidebar copy, and may replace `engine/src/jira.ts`. This branch only appends: new `engine/src/routines/` files; appended blocks in `types.ts`, `store.ts`, `orchestrator.ts` (one field + three one-line hooks), `server.ts` (two `case`s), `src/api.ts`, `src/mock.ts`, `messaging/service.ts` (one method at the end of the class), `src/views/Projects.tsx` (insert "Routines" into `STEPS` right before "Review" and add its step block), `src/views/Dashboard.tsx` (a card after the Merged section). Never reorder/reformat shared files; your branch rebases onto trackers before its PR.
- `npm run typecheck` and `npm run test:engine` pass after every task. Commit after every task.

## Review Focus

1. **Mac asleep past several scheduled times** — on wake, each overdue routine runs once, not once per missed slot. Pinned by the service unit test "overdue runs once" in Task 2.
2. **The agent prints prose around the envelope or no envelope at all** — the run is recorded `ok: false` with the tail of stdout, never posted. Pinned by Task 2 unit test.
3. **Cron edge cases** — `*/15`, `1-5` dow with `7`, month/day names, Feb 29, end-of-year rollover. Pinned by the cron tests in Task 1.
4. **A routine deleted while running** — the run completes, no crash, its state disappears afterwards. Pinned by Task 2 unit test.
5. **Two projects with routines due in the same tick plus the concurrency cap of 2** — the third waits for the next tick and is not lost. Pinned by Task 2 unit test.

```ts
// 5-field cron (min hour dom month dow), local time. Parsing never throws: it returns an Error for the UI.
export interface CronSpec {
  min: number[];
  hour: number[];
  dom: number[];
  month: number[];
  dow: number[];
  wild: { dom: boolean; dow: boolean };
}

const NAMES: Record<string, Record<string, number>> = {
  month: { jan: 1, feb: 2, mar: 3, apr: 4, may: 5, jun: 6, jul: 7, aug: 8, sep: 9, oct: 10, nov: 11, dec: 12 },
  dow: { sun: 0, mon: 1, tue: 2, wed: 3, thu: 4, fri: 5, sat: 6 },
};
const RANGES: Record<string, [number, number]> = { min: [0, 59], hour: [0, 23], dom: [1, 31], month: [1, 12], dow: [0, 7] };

function parseField(field: string, name: keyof typeof RANGES): number[] | Error {
  const [lo, hi] = RANGES[name];
  const out = new Set<number>();
  const num = (s: string) => {
    const n = NAMES[name]?.[s.toLowerCase()];
    if (n !== undefined) return n;
    if (!/^\d+$/.test(s)) return NaN;
    return Number(s);
  };
  for (const part of field.split(",")) {
    const [rangePart, stepPart] = part.split("/");
    const step = stepPart === undefined ? 1 : Number(stepPart);
    if (!Number.isInteger(step) || step < 1) return new Error(`bad step in “${part}”`);
    let a: number;
    let b: number;
    if (rangePart === "*") [a, b] = [lo, hi];
    else if (rangePart.includes("-")) {
      const [x, y] = rangePart.split("-");
      [a, b] = [num(x), num(y)];
    } else {
      a = num(rangePart);
      b = stepPart === undefined ? a : hi;
    }
    if (Number.isNaN(a) || Number.isNaN(b) || a < lo || b > hi || a > b) return new Error(`“${part}” is out of range for ${name} (${lo}-${hi})`);
    for (let v = a; v <= b; v += step) out.add(name === "dow" && v === 7 ? 0 : v);
  }
  return [...out].sort((x, y) => x - y);
}

export function parseCron(expr: string): CronSpec | Error {
  const fields = expr.trim().split(/\s+/);
  if (fields.length !== 5) return new Error("Use 5 fields: minute hour day-of-month month day-of-week");
  const names = ["min", "hour", "dom", "month", "dow"] as const;
  const parsed: Partial<CronSpec> = {};
  for (let i = 0; i < 5; i++) {
    const r = parseField(fields[i], names[i]);
    if (r instanceof Error) return r;
    (parsed as any)[names[i]] = r;
  }
  return { ...(parsed as Omit<CronSpec, "wild">), wild: { dom: fields[2] === "*", dow: fields[4] === "*" } };
}

/** Next matching minute strictly after `after`, local time. Error when nothing matches within 366 days. */
export function nextRun(spec: CronSpec, after: Date): Date | Error {
  const t = new Date(after.getTime());
  t.setSeconds(0, 0);
  t.setMinutes(t.getMinutes() + 1);
  const limit = after.getTime() + 366 * 86_400_000;
  while (t.getTime() <= limit) {
    if (!spec.month.includes(t.getMonth() + 1)) {
      t.setMonth(t.getMonth() + 1, 1);
      t.setHours(0, 0, 0, 0);
      continue;
    }
    const domOk = spec.dom.includes(t.getDate());
    const dowOk = spec.dow.includes(t.getDay());
    const dayOk = spec.wild.dom && spec.wild.dow ? true : spec.wild.dom ? dowOk : spec.wild.dow ? domOk : domOk || dowOk;
    if (!dayOk) {
      t.setDate(t.getDate() + 1);
      t.setHours(0, 0, 0, 0);
      continue;
    }
    if (!spec.hour.includes(t.getHours())) {
      t.setHours(t.getHours() + 1, 0, 0, 0);
      continue;
    }
    if (!spec.min.includes(t.getMinutes())) {
      t.setMinutes(t.getMinutes() + 1, 0, 0);
      continue;
    }
    return t;
  }
  return new Error("This schedule never matches a date in the next year.");
}

const two = (n: number) => String(n).padStart(2, "0");

export function describeCron(spec: CronSpec, raw: string): string {
  const times = spec.hour.length * spec.min.length <= 4 ? spec.hour.flatMap((h) => spec.min.map((m) => `${two(h)}:${two(m)}`)) : null;
  if (!times || !spec.wild.dom || spec.month.length !== 12) return raw;
  const list = times.length === 1 ? times[0] : `${times.slice(0, -1).join(", ")} and ${times[times.length - 1]}`;
  const dows = spec.dow.join(",");
  if (spec.wild.dow) return `every day at ${list}`;
  if (dows === "1,2,3,4,5") return `every weekday at ${list}`;
  if (dows === "0,6") return `every weekend day at ${list}`;
  return raw;
}
```

```ts
import type { Config, EngineState, MessageKind, ProjectConfig, RoutineConfig, RoutineRun, RoutineState } from "../types";
import { DEFAULT_ROUTINE_COMMAND } from "../types";
import { fill, sh, type RunOpts, type RunResult } from "../shell";
import { lastJson } from "../messaging/agent";
import { fillTemplate } from "../messaging/templates";
import { nextRun, parseCron } from "./cron";

export interface RoutineDeps {
  config: () => Config;
  state: () => EngineState;
  persist: () => void;
  log: (routineId: string, chunk: string) => void;
  notify: (title: string, body: string) => void;
  messaging: { postToProject(p: ProjectConfig, text: string, kind: MessageKind, idempotencyKey: string): Promise<{ id?: string }> };
  run?: (cmd: string, opts?: RunOpts) => Promise<RunResult>;
  now?: () => Date;
  maxConcurrent?: number;
}

const TICK_MS = 30_000;

export class RoutineService {
  private timer?: NodeJS.Timeout;
  private running = new Set<string>();
  constructor(public deps: RoutineDeps) {}

  private now() {
    return (this.deps.now || (() => new Date()))();
  }
  private all(): { p: ProjectConfig; r: RoutineConfig }[] {
    return this.deps.config().projects.flatMap((p) => p.routines.map((r) => ({ p, r })));
  }
  private stateFor(id: string): RoutineState {
    const s = this.deps.state();
    return (s.routines[id] ||= {});
  }

  start() {
    for (const st of Object.values(this.deps.state().routines)) {
      if (st.running) {
        st.running = false;
        if (st.last && !st.last.endedAt) Object.assign(st.last, { endedAt: this.now().toISOString(), ok: false, output: "HandsFree was quit" });
      }
    }
    this.configChanged();
    this.timer = setInterval(() => void this.tick(), TICK_MS);
  }

  shutdown() {
    clearInterval(this.timer);
  }

  /** Recompute schedules for every routine; drop state for routines that no longer exist and are not running. */
  configChanged() {
    const s = this.deps.state();
    const ids = new Set<string>();
    for (const { r } of this.all()) {
      ids.add(r.id);
      const st = this.stateFor(r.id);
      const spec = r.enabled ? parseCron(r.schedule) : undefined;
      if (!spec || spec instanceof Error) {
        st.nextRunAt = undefined;
        continue;
      }
      if (!st.nextRunAt || new Date(st.nextRunAt).getTime() <= 0) {
        const n = nextRun(spec, this.now());
        st.nextRunAt = n instanceof Error ? undefined : n.toISOString();
      } else {
        // Keep an existing (possibly overdue) slot, but make sure it matches the current expression.
        const n = nextRun(spec, new Date(new Date(st.nextRunAt).getTime() - 60_000));
        st.nextRunAt = n instanceof Error ? undefined : n.toISOString();
      }
    }
    for (const id of Object.keys(s.routines)) if (!ids.has(id) && !this.running.has(id)) delete s.routines[id];
    this.deps.persist();
  }

  async tick(now = this.now()) {
    const max = this.deps.maxConcurrent ?? 2;
    const due = this.all().filter(({ r }) => {
      const st = this.deps.state().routines[r.id];
      return r.enabled && st?.nextRunAt && new Date(st.nextRunAt) <= now && !this.running.has(r.id);
    });
    const runs: Promise<void>[] = [];
    for (const { p, r } of due) {
      if (this.running.size >= max) break; // the rest wait for the next tick; their nextRunAt stays in the past
      runs.push(this.execute(p, r, now));
    }
    await Promise.all(runs);
  }

  async runNow(id: string) {
    const hit = this.all().find(({ r }) => r.id === id);
    if (!hit) throw new Error("Unknown routine");
    if (this.running.has(id)) throw new Error("This routine is already running.");
    await this.execute(hit.p, hit.r, this.now());
  }

  readLog(id: string, max = 200_000) {
    return this.deps.state().routines[id] ? this.deps.log.length >= 0 && "" : ""; // replaced by the orchestrator (reads the file); kept for the interface
  }

  private async execute(p: ProjectConfig, r: RoutineConfig, now: Date) {
    const st = this.stateFor(r.id);
    const run: RoutineRun = { startedAt: now.toISOString() };
    st.last = run;
    st.running = true;
    this.running.add(r.id);
    // Schedule the next slot now, so an overdue routine runs once and moves on.
    const spec = parseCron(r.schedule);
    const n = spec instanceof Error ? spec : nextRun(spec, now);
    st.nextRunAt = n instanceof Error ? undefined : n.toISOString();
    this.deps.persist();
    const lastOk = Object.values(this.deps.state().routines).length ? st.last : undefined; // placeholder for type narrowing; real value below
    void lastOk;
    const prevOk = this.lastSuccessfulStart(r.id, run);
    const vars = {
      project: p.name,
      repo: repoSlug(p),
      repo_path: p.repoPath,
      channel: p.messaging.channelName || p.messaging.channelId,
      last_run: prevOk ? new Date(prevOk).toLocaleString() : "the beginning",
      now: now.toISOString(),
    };
    const prompt = fillTemplate(r.prompt, vars);
    const cmd = fill(r.agentCommand.trim() || DEFAULT_ROUTINE_COMMAND, { prompt });
    this.deps.log(r.id, `\n▶ ${r.name} · ${run.startedAt}\n$ ${cmd.slice(0, 300)}\n`);
    try {
      const res = await (this.deps.run || sh)(cmd, { cwd: p.repoPath || undefined, timeoutMs: r.timeoutMin * 60_000, env: { HANDSFREE: "1", HANDSFREE_ROUTINE: r.id }, onData: (d) => this.deps.log(r.id, d) });
      if (res.timedOut) throw new Error(`Timed out after ${r.timeoutMin} min`);
      if (res.code !== 0) throw new Error(`The agent exited with exit ${res.code}: ${(res.stderr || res.stdout).trim().slice(-300)}`);
      const env = lastJson(res.stdout || "");
      if (!env || typeof env !== "object" || !("result" in env)) throw new Error(`The agent did not print a result envelope: ${(res.stdout || "").trim().slice(-300)}`);
      if (env.is_error) throw new Error(String(env.result).slice(-400));
      run.output = String(env.result ?? "").trim().slice(0, 4000);
      run.ok = true;
      this.deps.log(r.id, `\n✓ ${r.name} finished\n${run.output}\n`);
      await this.post(p, r, run);
    } catch (e) {
      run.ok = false;
      run.output = String((e as Error).message || e).slice(0, 4000);
      this.deps.log(r.id, `\n✖ ${run.output}\n`);
      this.deps.notify(`${r.name} failed`, run.output.slice(0, 200));
    } finally {
      run.endedAt = this.now().toISOString();
      st.running = false;
      this.running.delete(r.id);
      // The routine may have been deleted while running.
      if (!this.all().some((x) => x.r.id === r.id)) delete this.deps.state().routines[r.id];
      this.deps.persist();
    }
  }

  private lastSuccessfulStart(id: string, current: RoutineRun) {
    const st = this.deps.state().routines[id];
    const prev = st?.lastOk;
    return prev && prev !== current.startedAt ? prev : undefined;
  }

  private async post(p: ProjectConfig, r: RoutineConfig, run: RoutineRun) {
    const st = this.stateFor(r.id);
    st.lastOk = run.startedAt;
    if (!r.postToChannel) return;
    if (p.messaging.platform === "none" || !p.messaging.channelId) {
      run.posted = false;
      run.postError = "no messaging platform or channel configured for this project";
      return;
    }
    try {
      await this.deps.messaging.postToProject(p, run.output || "", "routine", `routine:${r.id}:${run.startedAt}`);
      run.posted = true;
    } catch (e) {
      run.posted = false;
      run.postError = String((e as Error).message || e).slice(0, 400);
      this.deps.notify(`${r.name}: report not posted`, run.postError);
    }
  }
}

/** owner/name for github remotes, else the remote URL, else the repo path. */
export function repoSlug(p: Pick<ProjectConfig, "repoPath" | "remote">): string {
  // Filled by the orchestrator at run time when git is available; the pure fallback keeps tests deterministic.
  return p.repoPath || "";
}
```
